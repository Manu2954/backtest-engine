- Memory: 250 rows × 20 columns × 8 bytes = ~40KB (negligible!)

import asyncio
import json
from datetime import datetime
import websockets
import pandas as pd

class BinanceWebSocketFeed:
    """Real-time WebSocket feed for Binance crypto pairs."""

    def __init__(self, symbol: str, interval: str = "1m"):
        """
        Args:
            symbol: Trading pair (e.g., "btcusdt")
            interval: Kline interval (1m, 3m, 5m, 15m, 30m, 1h, 4h, 1d)
        """
        self.symbol = symbol.lower()
        self.interval = interval
        self.url = f"wss://stream.binance.com:9443/ws/{self.symbol}@kline_{interval}"
        self.ws = None
        self.callbacks = []  # List of async functions to call on new bar

    def subscribe(self, callback):
        """Register callback function to receive bars."""
        self.callbacks.append(callback)

    async def connect(self):
        """Connect to Binance WebSocket stream."""
        self.ws = await websockets.connect(self.url)
        print(f"Connected to {self.url}")

    async def start(self):
        """Start receiving bars (blocking loop)."""
        await self.connect()

        try:
            async for message in self.ws:
                data = json.loads(message)

                # Binance sends updates during the bar
                # Only process when bar is closed
                if data['k']['x']:  # x = is_closed
                    bar = self._parse_bar(data['k'])

                    # Call all registered callbacks
                    for callback in self.callbacks:
                        await callback(bar)

        except websockets.exceptions.ConnectionClosed:
            print("WebSocket closed, reconnecting...")
            await asyncio.sleep(5)
            await self.start()  # Reconnect

    def _parse_bar(self, kline) -> dict:
        """Parse Binance kline to OHLCV dict."""
        return {
            'timestamp': datetime.fromtimestamp(kline['t'] / 1000),
            'open': float(kline['o']),
            'high': float(kline['h']),
            'low': float(kline['l']),
            'close': float(kline['c']),
            'volume': float(kline['v']),
        }

import pandas as pd
from app.engine.indicator_layer import compute_indicators

class LiveIndicatorEngine:
    """Maintains rolling buffer of bars and computes indicators."""

    def __init__(self, strategy, buffer_size: int = 250):
        self.strategy = strategy
        self.buffer_size = buffer_size
        self.buffer = None  # Will be initialized with historical data

    async def initialize(self, historical_bars: pd.DataFrame):
        """Initialize buffer with historical data."""
        self.buffer = historical_bars.tail(self.buffer_size).copy()

        # Compute all indicators on initial buffer
        self.buffer = compute_indicators(
            self.buffer,
            self.strategy.indicators
        )

        print(f"Initialized buffer with {len(self.buffer)} bars")

    async def on_bar(self, bar: dict) -> pd.Series:
        """
        Process new bar and return updated indicators.

        Returns:
            Series with all indicator values for this bar
        """
        # Convert bar dict to DataFrame row
        new_row = pd.DataFrame([bar]).set_index('timestamp')

        # Append to buffer
        self.buffer = pd.concat([self.buffer, new_row]).tail(self.buffer_size)

        # Recompute indicators (only on buffer, not full history)
        self.buffer = compute_indicators(
            self.buffer,
            self.strategy.indicators
        )

        # Return latest row (current bar with indicators)
        return self.buffer.iloc[-1]

from dataclasses import dataclass
from datetime import datetime
import uuid

@dataclass
class Order:
    id: str
    ticker: str
    side: str  # "BUY" or "SELL"
    quantity: float
    order_type: str  # "MARKET" or "LIMIT"
    status: str  # "PENDING" | "FILLED" | "REJECTED"
    filled_price: float | None = None
    filled_at: datetime | None = None


class BinancePaperBroker:
    """Simulated broker for paper trading."""

    def __init__(self, initial_balance: float = 10000.0):
        self.balance_usdt = initial_balance
        self.positions = {}  # {ticker: quantity}
        self.orders = {}  # {order_id: Order}

    async def place_market_order(
        self,
        ticker: str,
        side: str,
        quantity: float,
        current_price: float,
        slippage_pct: float = 0.05
    ) -> Order:
        """
        Simulate market order fill.

        Args:
            ticker: Trading pair (e.g., "BTCUSDT")
            side: "BUY" or "SELL"
            quantity: Amount of base asset (e.g., 0.01 BTC)
            current_price: Current market price
            slippage_pct: Simulated slippage (0.05 = 0.05%)

        Returns:
            Order object with filled status
        """
        order = Order(
            id=str(uuid.uuid4()),
            ticker=ticker,
            side=side,
            quantity=quantity,
            order_type="MARKET",
            status="PENDING"
        )

        # Simulate slippage
        if side == "BUY":
            fill_price = current_price * (1 + slippage_pct / 100)
        else:
            fill_price = current_price * (1 - slippage_pct / 100)

        # Simulate commission (0.1% for Binance)
        commission_pct = 0.1

        if side == "BUY":
            cost = quantity * fill_price
            commission = cost * (commission_pct / 100)
            total_cost = cost + commission

            if total_cost > self.balance_usdt:
                order.status = "REJECTED"
                return order

            # Execute
            self.balance_usdt -= total_cost
            self.positions[ticker] = self.positions.get(ticker, 0) + quantity

        else:  # SELL
            current_position = self.positions.get(ticker, 0)
            if quantity > current_position:
                order.status = "REJECTED"
                return order

            # Execute
            proceeds = quantity * fill_price
            commission = proceeds * (commission_pct / 100)
            net_proceeds = proceeds - commission

            self.balance_usdt += net_proceeds
            self.positions[ticker] -= quantity

        order.status = "FILLED"
        order.filled_price = fill_price
        order.filled_at = datetime.utcnow()

        self.orders[order.id] = order
        return order

    def get_equity(self, current_prices: dict[str, float]) -> float:
        """Calculate total equity (cash + position value)."""
        position_value = sum(
            qty * current_prices.get(ticker, 0)
            for ticker, qty in self.positions.items()
        )
        return self.balance_usdt + position_value

from datetime import datetime
import asyncio
from app.engine.condition_engine import evaluate_conditions, evaluate_expression
from app.brokers.paper_broker import BinancePaperBroker

class TradingSession:
    """
    Manages a live trading session for a single ticker/strategy.
    """

    def __init__(
        self,
        session_id: str,
        strategy,
        ticker: str,
        broker: BinancePaperBroker,
        config: dict
    ):
        self.session_id = session_id
        self.strategy = strategy
        self.ticker = ticker
        self.broker = broker
        self.config = config

        # State
        self.state = "NO_POSITION"  # NO_POSITION | IN_POSITION
        self.current_position = None
        self.entry_price = None
        self.entry_time = None
        self.trades = []

        # Indicator engine
        self.indicator_engine = None

        # Lock for concurrent bar processing
        self.lock = asyncio.Lock()

    async def initialize(self, historical_bars):
        """Initialize with historical data for indicator warmup."""
        from app.realtime.live_indicator_engine import LiveIndicatorEngine

        self.indicator_engine = LiveIndicatorEngine(self.strategy)
        await self.indicator_engine.initialize(historical_bars)

    async def on_bar(self, bar: dict):
        """
        Process new bar: update indicators, check signals, execute trades.
        """
        async with self.lock:  # Prevent concurrent processing
            # Update indicators
            current_data = await self.indicator_engine.on_bar(bar)
            current_price = bar['close']

            # Check entry/exit signals
            if self.state == "NO_POSITION":
                if self._check_entry_signal(current_data):
                    await self._enter_position(current_price, bar['timestamp'])

            elif self.state == "IN_POSITION":
                exit_reason = self._check_exit_signal(current_data, current_price)
                if exit_reason:
                    await self._exit_position(
                        current_price,
                        bar['timestamp'],
                        exit_reason
                    )

    def _check_entry_signal(self, current_data: pd.Series) -> bool:
        """Evaluate entry conditions on current bar."""
        # Get full buffer (for lookback conditions)
        df = self.indicator_engine.buffer

        if self.strategy.entry_expression:
            # Use boolean expression
            entry_groups = self._build_groups_dict("ENTRY")
            signal_series = evaluate_expression(
                df,
                entry_groups,
                self.strategy.entry_expression
            )
            return signal_series.iloc[-1]  # Latest value
        else:
            # Legacy single group
            entry_group = self._group_to_payload("ENTRY")
            signal_series = evaluate_conditions(df, entry_group)
            return signal_series.iloc[-1]

    def _check_exit_signal(
        self,
        current_data: pd.Series,
        current_price: float
    ) -> str | None:
        """
        Check if exit conditions are met.

        Returns:
            Exit reason string if should exit, None otherwise
        """
        df = self.indicator_engine.buffer

        # 1. Check exit signal
        if self.strategy.exit_expression:
            exit_groups = self._build_groups_dict("EXIT")
            signal_series = evaluate_expression(
                df,
                exit_groups,
                self.strategy.exit_expression
            )
            if signal_series.iloc[-1]:
                return "signal"
        else:
            exit_group = self._group_to_payload("EXIT")
            signal_series = evaluate_conditions(df, exit_group)
            if signal_series.iloc[-1]:
                return "signal"

        # 2. Check stop loss
        if self.config.get('stop_loss_pct'):
            stop_price = self.entry_price * (1 - self.config['stop_loss_pct'] / 100)
            if current_price <= stop_price:
                return "stop_loss"

        # 3. Check take profit
        if self.config.get('take_profit_pct'):
            target_price = self.entry_price * (1 + self.config['take_profit_pct'] / 100)
            if current_price >= target_price:
                return "take_profit"

        # 4. Check dynamic stop (indicator-based)
        if self.config.get('dynamic_stop_column'):
            stop_col = self.config['dynamic_stop_column']
            if stop_col in current_data.index:
                stop_value = current_data[stop_col]
                if current_price <= stop_value:
                    return "trailing_stop"

        return None

    async def _enter_position(self, price: float, timestamp: datetime):
        """Enter a position at current price."""
        # Calculate position size
        equity = self.broker.get_equity({self.ticker: price})
        position_size_type = self.config.get('position_size_type', 'full_capital')
        position_size_value = self.config.get('position_size_value', 100.0)

        # Calculate quantity to buy
        if position_size_type == 'full_capital':
            amount_usdt = self.broker.balance_usdt
        elif position_size_type == 'percent_capital':
            amount_usdt = equity * (position_size_value / 100)
        elif position_size_type == 'fixed_amount':
            amount_usdt = position_size_value
        else:
            amount_usdt = self.broker.balance_usdt

        quantity = amount_usdt / price

        # Place order
        order = await self.broker.place_market_order(
            ticker=self.ticker,
            side="BUY",
            quantity=quantity,
            current_price=price,
            slippage_pct=self.config.get('slippage_pct', 0.05)
        )

        if order.status == "FILLED":
            self.state = "IN_POSITION"
            self.entry_price = order.filled_price
            self.entry_time = timestamp
            self.current_position = order.quantity

            print(f"[ENTRY] {timestamp} | "
                  f"Price: ${order.filled_price:.2f} | "
                  f"Qty: {order.quantity:.6f} | "
                  f"Cost: ${order.quantity * order.filled_price:.2f}")
        else:
            print(f"[ENTRY REJECTED] {order.id}")

    async def _exit_position(
        self,
        price: float,
        timestamp: datetime,
        reason: str
    ):
        """Exit current position at current price."""
        order = await self.broker.place_market_order(
            ticker=self.ticker,
            side="SELL",
            quantity=self.current_position,
            current_price=price,
            slippage_pct=self.config.get('slippage_pct', 0.05)
        )

        if order.status == "FILLED":
            # Calculate P&L
            pnl = (order.filled_price - self.entry_price) * order.quantity
            pnl_pct = ((order.filled_price / self.entry_price) - 1) * 100

            # Log trade
            trade = {
                'entry_time': self.entry_time,
                'entry_price': self.entry_price,
                'exit_time': timestamp,
                'exit_price': order.filled_price,
                'quantity': order.quantity,
                'pnl': pnl,
                'pnl_pct': pnl_pct,
                'exit_reason': reason
            }
            self.trades.append(trade)

            print(f"[EXIT - {reason.upper()}] {timestamp} | "
                  f"Price: ${order.filled_price:.2f} | "
                  f"P&L: ${pnl:.2f} ({pnl_pct:+.2f}%) | "
                  f"Balance: ${self.broker.balance_usdt:.2f}")

            # Reset state
            self.state = "NO_POSITION"
            self.current_position = None
            self.entry_price = None
            self.entry_time = None
        else:
            print(f"[EXIT REJECTED] {order.id}")

from datetime import datetime
import httpx

class BinanceLiveBroker:
    """
    Real Binance broker for live trading.

    REQUIRES:
    - Binance API Key (read + trade permissions)
    - Binance Secret Key
    - IP whitelisting recommended
    """

    BASE_URL = "https://api.binance.com"

    def __init__(self, api_key: str, secret_key: str):
        self.api_key = api_key
        self.secret_key = secret_key

    def _sign_request(self, params: dict) -> dict:
        """Sign request parameters with HMAC SHA256."""
        params['timestamp'] = int(time.time() * 1000)
        query_string = '&'.join([f"{k}={v}" for k, v in params.items()])
        signature = hmac.new(
            self.secret_key.encode(),
            query_string.encode(),
            hashlib.sha256
        ).hexdigest()
        params['signature'] = signature
        return params

    async def place_market_order(
        self,
        ticker: str,
        side: str,
        quantity: float
    ):
        """
        Place REAL market order on Binance.

        Args:
            ticker: Symbol (e.g., "BTCUSDT")
            side: "BUY" or "SELL"
            quantity: Base asset quantity (e.g., 0.001 BTC)
        """
        params = {
            'symbol': ticker,
            'side': side,
            'type': 'MARKET',
            'quantity': quantity
        }
        params = self._sign_request(params)

        async with httpx.AsyncClient() as client:
            response = await client.post(
                f"{self.BASE_URL}/api/v3/order",
                params=params,
                headers={'X-MBX-APIKEY': self.api_key}
            )
            response.raise_for_status()
            return response.json()

    async def get_balance(self) -> dict:
        """Get account balances."""
        params = self._sign_request({})

        async with httpx.AsyncClient() as client:
            response = await client.get(
                f"{self.BASE_URL}/api/v3/account",
                params=params,
                headers={'X-MBX-APIKEY': self.api_key}
            )
            response.raise_for_status()

            balances = {}
            for balance in response.json()['balances']:
                asset = balance['asset']
                free = float(balance['free'])
                locked = float(balance['locked'])
                if free > 0 or locked > 0:
                    balances[asset] = {'free': free, 'locked': locked}

            return balances

class RiskManager:
    """Enforce risk limits on live trading."""

    def __init__(self, config: dict):
        self.config = config
        self.daily_pnl = 0.0
        self.daily_trades = 0
        self.consecutive_losses = 0
        self.max_drawdown_hit = False

    def check_can_trade(
        self,
        current_equity: float,
        initial_equity: float
    ) -> tuple[bool, str]:
        """
        Validate if trading is allowed.

        Returns:
            (allowed: bool, reason: str)
        """
        # 1. Max daily loss
        max_daily_loss = self.config.get('max_daily_loss_pct', 5.0)
        daily_loss_pct = (self.daily_pnl / initial_equity) * 100
        if daily_loss_pct < -max_daily_loss:
            return False, f"Daily loss limit hit: {daily_loss_pct:.2f}%"

        # 2. Max trades per day
        max_daily_trades = self.config.get('max_daily_trades', 10)
        if self.daily_trades >= max_daily_trades:
            return False, f"Daily trade limit hit: {self.daily_trades}"

        # 3. Max consecutive losses
        max_losses = self.config.get('max_consecutive_losses', 3)
        if self.consecutive_losses >= max_losses:
            return False, f"Consecutive loss limit: {self.consecutive_losses}"

        # 4. Max drawdown
        max_dd = self.config.get('max_drawdown_pct', 10.0)
        current_dd = ((current_equity / initial_equity) - 1) * 100
        if current_dd < -max_dd:
            self.max_drawdown_hit = True
            return False, f"Max drawdown exceeded: {current_dd:.2f}%"

        return True, "OK"
