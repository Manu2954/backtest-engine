import pandas as pd
import numpy as np

def detect_order_blocks(df: pd.DataFrame, lookback: int = 20) -> pd.DataFrame:
    """
    Detect order blocks (OB) - zones where institutions entered positions.

    Bullish OB: Last down candle before strong bullish move
    Bearish OB: Last up candle before strong bearish move

    Args:
        df: OHLCV DataFrame
        lookback: Bars to look back for structure

    Returns:
        DataFrame with added columns:
        - bullish_ob_high: High of bullish order block
        - bullish_ob_low: Low of bullish order block
        - bearish_ob_high: High of bearish order block
        - bearish_ob_low: Low of bearish order block
        - is_in_bullish_ob: True if price is touching bullish OB
        - is_in_bearish_ob: True if price is touching bearish OB
    """
    df = df.copy()

    # Initialize columns
    df['bullish_ob_high'] = np.nan
    df['bullish_ob_low'] = np.nan
    df['bearish_ob_high'] = np.nan
    df['bearish_ob_low'] = np.nan

    for i in range(lookback, len(df)):
        # Bullish OB: Find last red candle before big green move
        # Look back for: down close -> strong up move (>2% in next 3 bars)
        if df['close'].iloc[i-1] < df['open'].iloc[i-1]:  # Red candle
            # Check if followed by strong bullish move
            future_high = df['high'].iloc[i:i+3].max()
            if future_high > df['close'].iloc[i-1] * 1.02:  # 2% up
                # Mark the red candle as bullish OB
                df.loc[df.index[i-1], 'bullish_ob_high'] = df['high'].iloc[i-1]
                df.loc[df.index[i-1], 'bullish_ob_low'] = df['low'].iloc[i-1]

        # Bearish OB: Find last green candle before big red move
        if df['close'].iloc[i-1] > df['open'].iloc[i-1]:  # Green candle
            future_low = df['low'].iloc[i:i+3].min()
            if future_low < df['close'].iloc[i-1] * 0.98:  # 2% down
                df.loc[df.index[i-1], 'bearish_ob_high'] = df['high'].iloc[i-1]
                df.loc[df.index[i-1], 'bearish_ob_low'] = df['low'].iloc[i-1]

    # Forward fill OB levels (they remain valid until price revisits)
    df['bullish_ob_high'] = df['bullish_ob_high'].fillna(method='ffill')
    df['bullish_ob_low'] = df['bullish_ob_low'].fillna(method='ffill')
    df['bearish_ob_high'] = df['bearish_ob_high'].fillna(method='ffill')
    df['bearish_ob_low'] = df['bearish_ob_low'].fillna(method='ffill')

    # Check if current price is touching OB
    df['is_in_bullish_ob'] = (
        (df['low'] <= df['bullish_ob_high']) &
        (df['low'] >= df['bullish_ob_low'])
    )
    df['is_in_bearish_ob'] = (
        (df['high'] >= df['bearish_ob_low']) &
        (df['high'] <= df['bearish_ob_high'])
    )

    return df


def detect_fair_value_gaps(df: pd.DataFrame) -> pd.DataFrame:
    """
    Detect Fair Value Gaps (FVG) / Imbalances.

    Bullish FVG: Gap between bar[i].high and bar[i+2].low
    Bearish FVG: Gap between bar[i].low and bar[i+2].high

    Args:
        df: OHLCV DataFrame

    Returns:
        DataFrame with added columns:
        - bullish_fvg_high: Top of bullish FVG
        - bullish_fvg_low: Bottom of bullish FVG
        - bearish_fvg_high: Top of bearish FVG
        - bearish_fvg_low: Bottom of bearish FVG
        - is_in_bullish_fvg: True if price is in bullish FVG
        - is_in_bearish_fvg: True if price is in bearish FVG
    """
    df = df.copy()

    df['bullish_fvg_high'] = np.nan
    df['bullish_fvg_low'] = np.nan
    df['bearish_fvg_high'] = np.nan
    df['bearish_fvg_low'] = np.nan

    for i in range(2, len(df)):
        # Bullish FVG: Gap between bar[i-2].high and bar[i].low
        # Middle bar (i-1) doesn't fill the gap
        gap_bottom = df['high'].iloc[i-2]
        gap_top = df['low'].iloc[i]

        if gap_top > gap_bottom:  # There's a gap
            # Check if middle bar doesn't fill it
            if df['high'].iloc[i-1] < gap_top and df['low'].iloc[i-1] > gap_bottom:
                df.loc[df.index[i], 'bullish_fvg_high'] = gap_top
                df.loc[df.index[i], 'bullish_fvg_low'] = gap_bottom

        # Bearish FVG: Gap between bar[i-2].low and bar[i].high
        gap_top = df['low'].iloc[i-2]
        gap_bottom = df['high'].iloc[i]

        if gap_bottom < gap_top:  # There's a gap
            if df['low'].iloc[i-1] > gap_bottom and df['high'].iloc[i-1] < gap_top:
                df.loc[df.index[i], 'bearish_fvg_high'] = gap_top
                df.loc[df.index[i], 'bearish_fvg_low'] = gap_bottom

    # Forward fill FVGs until price fills them
    df['bullish_fvg_high'] = df['bullish_fvg_high'].fillna(method='ffill')
    df['bullish_fvg_low'] = df['bullish_fvg_low'].fillna(method='ffill')
    df['bearish_fvg_high'] = df['bearish_fvg_high'].fillna(method='ffill')
    df['bearish_fvg_low'] = df['bearish_fvg_low'].fillna(method='ffill')

    # Check if current price is in FVG
    df['is_in_bullish_fvg'] = (
        (df['close'] >= df['bullish_fvg_low']) &
        (df['close'] <= df['bullish_fvg_high'])
    )
    df['is_in_bearish_fvg'] = (
        (df['close'] >= df['bearish_fvg_low']) &
        (df['close'] <= df['bearish_fvg_high'])
    )

    return df


def detect_liquidity_pools(df: pd.DataFrame, lookback: int = 50) -> pd.DataFrame:
    """
    Detect liquidity pools - areas where stop losses cluster.

    Buy-side liquidity: Above recent swing highs (buy stops sit here)
    Sell-side liquidity: Below recent swing lows (sell stops sit here)

    Args:
        df: OHLCV DataFrame
        lookback: Period to find swing points

    Returns:
        DataFrame with added columns:
        - buy_side_liquidity: Price level above which buy stops cluster
        - sell_side_liquidity: Price level below which sell stops cluster
    """
    df = df.copy()

    # Find swing highs (local peaks)
    df['swing_high'] = df['high'].rolling(lookback, center=True).apply(
        lambda x: x[len(x)//2] if x[len(x)//2] == x.max() else np.nan
    )

    # Find swing lows (local troughs)
    df['swing_low'] = df['low'].rolling(lookback, center=True).apply(
        lambda x: x[len(x)//2] if x[len(x)//2] == x.min() else np.nan
    )

    # Buy-side liquidity: Just above recent swing high
    df['buy_side_liquidity'] = df['swing_high'].rolling(lookback).max() * 1.001

    # Sell-side liquidity: Just below recent swing low
    df['sell_side_liquidity'] = df['swing_low'].rolling(lookback).min() * 0.999

    return df
