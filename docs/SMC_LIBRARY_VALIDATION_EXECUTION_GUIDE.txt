What to Mark:
-------------

1. FAIR VALUE GAPS (FVGs)
   Mark each FVG with:
   - Row index (bar number)
   - FVG type: "bullish" or "bearish"
   - Top price level
   - Bottom price level
   - Notes: Why you marked it (e.g., "significant gap, unfilled")

2. ORDER BLOCKS (OBs)
   Mark each OB with:
   - Row index
   - OB type: "bullish" or "bearish"
   - Top price (high of OB candle)
   - Bottom price (low of OB candle)
   - Strength: "strong" or "weak" (subjective)
   - Notes: Why you marked it (e.g., "last red candle before 5% pump")

3. SWING HIGHS/LOWS
   Mark each swing with:
   - Row index
   - Type: "swing_high" or "swing_low"
   - Price level
   - Notes: Lookback period you used (e.g., "10 bars each side")

4. BREAK OF STRUCTURE (BOS)
   Mark each BOS with:
   - Row index (bar where break occurred)
   - Type: "bullish_bos" or "bearish_bos"
   - Previous swing level that was broken
   - Notes: Close vs wick break

5. LIQUIDITY POOLS
   Mark each pool with:
   - Row index (where you first identify it)
   - Type: "buy_side" (above) or "sell_side" (below)
   - Price level (where stops cluster)
   - Notes: Why you think stops are here


Documentation Format:
---------------------

Create CSV file per chart: "btc_1h_chart_1_labels.csv"

Columns:
  - bar_index: int (row number in OHLCV data)
  - timestamp: datetime (from original data)
  - concept: string (FVG, OB, SWING_HIGH, etc.)
  - type: string (bullish, bearish, buy_side, etc.)
  - price_high: float (top of zone)
  - price_low: float (bottom of zone)
  - strength: string (strong, weak, N/A)
  - notes: string (why you marked it)

Example:
bar_index,timestamp,concept,type,price_high,price_low,strength,notes
45,2024-01-15 09:00,FVG,bullish,43500.0,43200.0,N/A,Significant gap unfilled
48,2024-01-15 12:00,OB,bullish,43100.0,42900.0,strong,Last red before 5% pump
52,2024-01-15 16:00,SWING_LOW,swing_low,42800.0,42800.0,N/A,10 bars each side

"""
SMC Library Validation Script
Runs smartmoneyconcepts library on labeled charts and compares to manual labels.
"""
import pandas as pd
from smartmoneyconcepts import smc
from pathlib import Path
import sys

def load_chart_data(chart_file: str) -> pd.DataFrame:
    """Load OHLCV chart data."""
    df = pd.read_csv(chart_file, index_col=0, parse_dates=True)

    # Library requires lowercase columns
    df.columns = df.columns.str.lower()

    return df

def load_manual_labels(labels_file: str) -> pd.DataFrame:
    """Load manual labels CSV."""
    labels = pd.read_csv(labels_file)
    labels['timestamp'] = pd.to_datetime(labels['timestamp'])
    return labels

def detect_with_library(df: pd.DataFrame, params: dict) -> pd.DataFrame:
    """Run library detection on chart."""

    # Detect FVGs
    df = smc.fvg(df, join_consecutive=False)

    # Detect Swings
    df = smc.swing_highs_lows(df, swing_length=params.get('swing_length', 50))

    # Detect Order Blocks
    df = smc.ob(df, swing_length=params.get('ob_swing_length', 20))

    # Detect BOS
    df = smc.bos(df, swing_length=params.get('bos_swing_length', 50), close_break=True)

    # Detect Liquidity
    df = smc.liquidity(df, range_percent=params.get('range_percent', 0.01))

    return df

def compare_fvg(df: pd.DataFrame, manual: pd.DataFrame) -> dict:
    """Compare FVG detection accuracy."""

    # Extract manual FVG labels
    manual_fvg_bullish = manual[
        (manual['concept'] == 'FVG') &
        (manual['type'] == 'bullish')
    ]['bar_index'].tolist()

    manual_fvg_bearish = manual[
        (manual['concept'] == 'FVG') &
        (manual['type'] == 'bearish')
    ]['bar_index'].tolist()

    # Extract library FVG detections
    library_fvg_bullish = df[df['FVG'].notna() & (df['FVG'] > 0)].index.tolist()
    library_fvg_bearish = df[df['FVG'].notna() & (df['FVG'] < 0)].index.tolist()

    # Calculate matches
    bullish_matches = len(set(manual_fvg_bullish) & set(library_fvg_bullish))
    bearish_matches = len(set(manual_fvg_bearish) & set(library_fvg_bearish))

    total_manual = len(manual_fvg_bullish) + len(manual_fvg_bearish)
    total_library = len(library_fvg_bullish) + len(library_fvg_bearish)
    total_matches = bullish_matches + bearish_matches

    # Calculate metrics
    precision = total_matches / total_library if total_library > 0 else 0
    recall = total_matches / total_manual if total_manual > 0 else 0
    f1_score = 2 * (precision * recall) / (precision + recall) if (precision + recall) > 0 else 0

    return {
        'manual_count': total_manual,
        'library_count': total_library,
        'matches': total_matches,
        'precision': precision * 100,
        'recall': recall * 100,
        'f1_score': f1_score * 100,
        'manual_bullish': manual_fvg_bullish,
        'library_bullish': library_fvg_bullish,
        'manual_bearish': manual_fvg_bearish,
        'library_bearish': library_fvg_bearish
    }

    """Validate library on single chart."""

    print(f"Validating: {chart_file}")

    # Load data
    df = load_chart_data(chart_file)

def main():
    """Run validation on all charts."""

    # Parameters to test
    params = {
        'swing_length': 50,
        'ob_swing_length': 20,
        'bos_swing_length': 50,
        'range_percent': 0.01
    }

    # Find all chart files
    data_dir = Path("validation_data")
    chart_files = sorted(data_dir.glob("*_chart_*.csv"))

    all_results = []

    for chart_file in chart_files:
        # Find corresponding labels file
        labels_file = chart_file.stem + "_labels.csv"
        labels_path = data_dir / labels_file

        if not labels_path.exists():
            print(f"Warning: No labels for {chart_file.name}")
            continue

        # Validate
        result = validate_chart(str(chart_file), str(labels_path), params)
        all_results.append(result)

    # Aggregate results
    print_summary(all_results)

    # Save detailed results
    save_results(all_results, "validation_results.json")

def print_summary(results: list):
    """Print accuracy summary."""

    print("\n" + "="*80)
    print("VALIDATION SUMMARY")
    print("="*80)

    # Calculate averages across all charts
    avg_fvg_f1 = sum(r['fvg']['f1_score'] for r in results) / len(results)
    avg_ob_f1 = sum(r['ob']['f1_score'] for r in results) / len(results)

    print(f"\nFair Value Gaps (FVG):")
    print(f"  Average F1 Score: {avg_fvg_f1:.1f}%")
    print(f"  Target: >75%")
    print(f"  Status: {'✅ PASS' if avg_fvg_f1 >= 75 else '❌ FAIL'}")

    print(f"  Status: {'✅ PASS' if avg_ob_f1 >= 65 else '❌ FAIL'}")

    print(f"\n" + "="*80)
    print(f"OVERALL AVERAGE: {overall_avg:.1f}%")

    else:

================================================================================
                    TROUBLESHOOTING
================================================================================

Problem: Library accuracy is 40-50% (very low)
Solution:
  - Check if your manual labels are consistent
  - Verify parameter settings (default may not match your style)
  - Review library documentation (may use different definitions)
  - Decision: Likely need custom implementation

Solution:
  - Library is too sensitive (low threshold)
  - Try tuning parameters to be more selective
  - Add filtering logic in wrapper (volume, size, etc.)

Problem: Library misses 60% of your patterns
Solution:
  - Library is too strict (high threshold)
  - Check if your criteria are more lenient
  - May need custom implementation with your criteria

Problem: Results inconsistent across charts
Solution:
  - May need different parameters per market condition
  - Adaptive parameter logic needed
  - Consider ML approach instead of fixed rules
